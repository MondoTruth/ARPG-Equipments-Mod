-	script	ARPG_Chest2_Drop	-1,{
	end;

OnInit:
	.id = 55001;
	.default_rate_low = callfunc("F_ModSetting", "arpg-equipments", "chest_2_rate", 5) * 100;
	.rate_secondary   = 200;  // 2% (Lvl 61 - 99)
	end;

OnNPCKillEvent:
	.@lv = getmonsterinfo(killedrid, MOB_LV);
	if (.@lv < 31 || .@lv > 99) end;

	.@main_rate = ($Chest2_Rate > 0) ? $Chest2_Rate : .default_rate_low;

	.@chance = 0;
	if (.@lv <= 60) {
		.@chance = .@main_rate;
	} else {
		.@chance = .rate_secondary;
	}

	if (.@chance > 0 && rand(10000) < .@chance) {
		if (checkweight(.id, 1)) {
			getitem .id, 1;
			dispbottom "You found a Tier 2 Equipment Chest!", 0x00FFFF;
		}
	}
	end;
}
