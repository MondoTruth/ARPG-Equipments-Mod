-	script	ARPG_Chest3_Drop	-1,{
	end;

OnInit:
	.id = 55002;
	.default_rate_low = callfunc("F_ModSetting", "arpg-equipments", "chest_3_rate", 3) * 100;
	.rate_secondary   = 200;  // 2% (Lvl 100 - 150)
	end;

OnNPCKillEvent:
	.@lv = getmonsterinfo(killedrid, MOB_LV);
	if (.@lv < 61 || .@lv > 150) end;

	// Use custom rate if configured, otherwise default to 300 (3%)
	.@main_rate = ($Chest3_Rate > 0) ? $Chest3_Rate : .default_rate_low;

	.@chance = 0;
	if (.@lv <= 99) {
		.@chance = .@main_rate;
	} else {
		.@chance = .rate_secondary;
	}

	if (.@chance > 0 && rand(10000) < .@chance) {
		if (checkweight(.id, 1)) {
			getitem .id, 1;
			dispbottom "You found a Tier 3 Equipment Chest!", 0x3399FF;
		}
	}
	end;
}
