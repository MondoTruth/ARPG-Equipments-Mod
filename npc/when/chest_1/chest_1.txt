-	script	ARPG_Chest1_Drop	-1,{
	end;

OnInit:
	.id = 55000;
	// Primary rate (Lvl 1 - 30) comes from the mod setting "chest_1_rate" (percent).
	// Convert to basis points (1 - 10000) to match $Chest1_Rate.
	.default_rate_low = callfunc("F_ModSetting", "arpg-equipments", "chest_1_rate", 5) * 100;
	.rate_secondary   = 200;  // 2% (Lvl 31 - 99)
	end;

OnNPCKillEvent:
	.@lv = getmonsterinfo(killedrid, MOB_LV);
	if (.@lv < 1 || .@lv > 99) end;

	// Use custom rate if configured, otherwise default to 500
	.@main_rate = ($Chest1_Rate > 0) ? $Chest1_Rate : .default_rate_low;

	.@chance = 0;
	if (.@lv <= 30) {
		.@chance = .@main_rate;
	} else {
		.@chance = .rate_secondary;
	}

	if (.@chance > 0 && rand(10000) < .@chance) {
		if (checkweight(.id, 1)) {
			getitem .id, 1;
			dispbottom "You found a Tier 1 Equipment Chest!", 0x00FF88;
		}
	}
	end;
}
